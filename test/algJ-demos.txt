/* ---------- LAB 157 포물선 — 위성 안테나가 접시 모양인 까닭 ---------- */
function paraFocusDemo(root){

    </div>

      ${slider('pfP','초점까지의 거리 p',0.5,3,0.1,1,'')}

    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.strokeStyle='#E5E7EB'; ctx.lineWidth=1;

    ctx.strokeStyle='#2563EB'; ctx.lineWidth=2.6; ctx.beginPath();

    const py=f(px);

/* ---------- LAB 160 거리 차로 위치 찾기 — 소리의 도착 시간 차 ---------- */
function tdoaDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="td1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">두 마이크에 소리가 <b>도착한 시간 차</b>가 같은 점들을 이으면
      쌍곡선이 됩니다. 마이크를 하나 더 놓으면?</div>
    </div>
    <div class="col">
      ${slider('tdD','시간 차 (마이크 A 가 먼저)',-8,8,0.2,0,'ms')}
      <div class="ctl" style="display:flex;gap:6px;flex-wrap:wrap;margin-top:8px">
        <button class="b g sm" id="tdThird">세 번째 마이크 켜기</button>
        <button class="b g sm" id="tdOff">끄기</button>
      </div>
      <div class="out" id="tdout"></div>
      <div class="note" style="margin-top:12px">거리 차 = 소리 속도 × 시간 차.
      <b>거리 차가 일정한 점의 자취</b>가 곧 쌍곡선입니다 — 이것이 위치 추적의 원리입니다.</div>
    </div></div>`;
  const cv=$('#td1'),ctx=cv.getContext('2d'),W=380,H=330;
  const S={ third:false };
  const V=0.343; // m/ms
  const A=[-4,0], B=[4,0], C=[0,5];
  function draw(){
    const dt=+$('#tdD').value; $('#tdDv').textContent=dt.toFixed(1)+'ms';
    const dd=V*dt*2; // 거리 차(m) — 화면 단위와 맞춘 스케일
    const X=x=>W/2+34*x, Y=y=>H-70-34*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.strokeStyle='#E5E7EB'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(10,Y(0)); ctx.lineTo(W-10,Y(0)); ctx.stroke();
    const drawHyp=(P1,P2,diff,col)=>{
      const cx=(P1[0]+P2[0])/2, cy=(P1[1]+P2[1])/2;
      const dx=P2[0]-P1[0], dy=P2[1]-P1[1], L=Math.hypot(dx,dy), c=L/2;
      const a=Math.abs(diff)/2; if(a>=c-0.02) return false;
      const b=Math.sqrt(c*c-a*a), ang=Math.atan2(dy,dx), sgn=diff>0?-1:1;
      ctx.strokeStyle=col; ctx.lineWidth=2.2; ctx.beginPath();
      for(let i=0;i<=120;i++){ const t=-2.2+4.4*i/120;
        const u=sgn*a*Math.cosh(t), v=b*Math.sinh(t);
        const xx=cx+u*Math.cos(ang)-v*Math.sin(ang), yy=cy+u*Math.sin(ang)+v*Math.cos(ang);
        i?ctx.lineTo(X(xx),Y(yy)):ctx.moveTo(X(xx),Y(yy)); }
      ctx.stroke(); return true; };
    const ok1=drawHyp(A,B,dd,'#2563EB');
    let ok2=false;
    if(S.third){ const src=[1.6,3.2];
      const d2=Math.hypot(src[0]-A[0],src[1]-A[1])-Math.hypot(src[0]-C[0],src[1]-C[1]);
      ok2=drawHyp(A,C,d2,'#0E9384');
      ctx.fillStyle='#BE123C'; ctx.beginPath(); ctx.arc(X(src[0]),Y(src[1]),5.4,0,7); ctx.fill();
      ctx.fillStyle='#BE123C'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
      ctx.fillText('소리가 난 곳', X(src[0])+8, Y(src[1])-6); }
    [[A,'A'],[B,'B']].concat(S.third?[[C,'C']]:[]).forEach(([P,nm])=>{
      ctx.fillStyle='#111827'; ctx.fillRect(X(P[0])-5,Y(P[1])-5,10,10);
      ctx.fillStyle='#374151'; ctx.font='11px sans-serif'; ctx.textAlign='center';
      ctx.fillText(nm, X(P[0]), Y(P[1])+20); });
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('마이크 사이 거리 8 m · 소리 속도 343 m/s', 12, H-12);
    $('#tdout').innerHTML =
      `<span class="mono">시간 차 ${dt.toFixed(1)} ms → 거리 차 = 0.343 × ${dt.toFixed(1)} × 2 = <b>${dd.toFixed(2)} m</b></span><br>`
      + (ok1 ? `<span class="mono">거리 차가 ${Math.abs(dd).toFixed(2)} m 인 점들이 파란 쌍곡선을 이룹니다.</span><br>` : `<b>거리 차가 마이크 간격보다 커서 그런 점은 없습니다.</b><br>`)
      + (ok1&&Math.abs(dd)>0.5&&!S.third ? `<b style="color:#B45309">아직 곡선 위 어디인지까지는 모릅니다 — 마이크가 둘뿐이기 때문입니다.</b><br>` : '')
      + (S.third&&ok2 ? `<b style="color:#7C3AED">두 쌍곡선이 만나 한 점으로 좁혀졌습니다 — 마이크 셋이면 위치가 정해집니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">GPS 도 같은 원리입니다 — 위성 신호의 도착 시간 차를 씁니다.</span>`;
  }
  $('#tdThird').onclick=()=>{ S.third=true; draw(); };
  $('#tdOff').onclick=()=>{ S.third=false; draw(); };
  $('#tdD').oninput=draw; draw();
  inquiry(root,{ id:'td', title:'거리 차로 찾는 위치',
    q:['시간 차를 바꾸며 (시간 차, 거리 차, 곡선의 모양) 을 다섯 줄 기록하세요.',
       '시간 차가 0 이면 어떤 도형이 되나요?',
       '마이크가 둘일 때와 셋일 때 알 수 있는 것의 차이를 쓰세요.'],
    cols:['시간 차(ms)','거리 차(m)','a','마이크 수','알 수 있는 것'],
    hint:'힌트 — 거리 차의 절반이 a 입니다. a 가 c 를 넘으면 그런 점은 없습니다.',
    read:()=>{ const dt=+$('#tdD').value, dd=0.343*dt*2;
      return [dt.toFixed(1),dd.toFixed(2),(Math.abs(dd)/2).toFixed(2),S.third?3:2,
        S.third?'한 점':'곡선 하나']; } });
}

/* ---------- LAB 161 접선의 조건 — 판별식이 0 이 되는 순간 ---------- */
function tangentCondDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="tc1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">직선을 위아래로 옮기며 원과 <b>몇 군데서 만나는지</b> 세어 보세요.
      두 점 → 한 점 → 만나지 않음으로 바뀝니다.</div>
    </div>
    <div class="col">
      ${slider('tcM','직선의 기울기 m',-2,2,0.1,0.8,'')}
      ${slider('tcK','직선의 y 절편 k',-6,6,0.1,1,'')}
      <div class="ctl" style="margin-top:8px"><button class="b g sm" id="tcFit">접하도록 k 맞추기</button></div>
      <div class="out" id="tcout"></div>
      <div class="note" style="margin-top:12px">x² + y² = 4 와 y = mx + k 를 연립하면 이차방정식.
      <b>판별식 D 의 부호</b>가 교점의 개수를 정합니다.</div>
    </div></div>`;
  const cv=$('#tc1'),ctx=cv.getContext('2d'),W=380,H=320;
  const R=2;
  function draw(){
    const m=+$('#tcM').value, k=+$('#tcK').value;
    $('#tcMv').textContent=m.toFixed(1); $('#tcKv').textContent=k.toFixed(1);
    const X=x=>W/2+52*x, Y=y=>H/2-52*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.strokeStyle='#E5E7EB'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(10,Y(0)); ctx.lineTo(W-10,Y(0)); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(X(0),10); ctx.lineTo(X(0),H-10); ctx.stroke();
    ctx.strokeStyle='#2563EB'; ctx.lineWidth=2.4;
    ctx.beginPath(); ctx.arc(X(0),Y(0),52*R,0,7); ctx.stroke();
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=2.2;
    ctx.beginPath(); ctx.moveTo(X(-3.4),Y(m*(-3.4)+k)); ctx.lineTo(X(3.4),Y(m*3.4+k)); ctx.stroke();
    const A=1+m*m, B=2*m*k, C=k*k-R*R, D=B*B-4*A*C;
    if(D>=0){ [(-B+Math.sqrt(D))/(2*A), (-B-Math.sqrt(D))/(2*A)].forEach(x=>{
      ctx.fillStyle='#0E9384'; ctx.beginPath(); ctx.arc(X(x),Y(m*x+k),5,0,7); ctx.fill(); }); }
    const dist=Math.abs(k)/Math.sqrt(1+m*m);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('원 x² + y² = 4 (반지름 2) · 중심에서 직선까지 '+dist.toFixed(3), 12, H-12);
    const tol=0.01;
    const n = D>tol ? 2 : (D<-tol ? 0 : 1);
    $('#tcout').innerHTML =
      `<span class="mono">(1+m²)x² + 2mkx + (k²−4) = 0 · D = <b>${D.toFixed(4)}</b></span><br>`
      + `<span class="mono">교점 <b>${n}</b>개 · 중심까지 거리 ${dist.toFixed(3)} · 반지름 2</span><br>`
      + (n===1 ? `<b style="color:#B45309">판별식이 0 — 한 점에서만 스치듯 만납니다. 접선입니다.</b><br>` : '')
      + (n===0 ? `<b style="color:#7C3AED">판별식이 음수 — 실근이 없어 아예 만나지 않습니다.</b><br>` : '')
      + (n===2 ? `두 점에서 만납니다 — 할선입니다.<br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">「중심까지 거리 = 반지름」과 「D = 0」이 같은 말임을 확인하세요.</span>`;
  }
  $('#tcFit').onclick=()=>{ const m=+$('#tcM').value;
    $('#tcK').step=0.0001; $('#tcK').value=Math.sqrt(4*(1+m*m)); draw(); };
  ['tcM','tcK'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'tc', title:'접선의 조건 — D = 0',
    q:['k 를 바꾸며 (k, D, 교점 수, 중심까지 거리) 를 여섯 줄 기록하세요.',
       '교점이 2 → 1 → 0 으로 바뀌는 경계의 k 를 찾으세요.',
       '「중심까지 거리 = 반지름」과 「D = 0」이 같은 조건임을 식으로 보이세요.'],
    cols:['m','k','D','교점 수','중심까지 거리'],
    hint:'힌트 — D = 0 을 k 에 대해 풀면 k² = 4(1 + m²) 입니다.',
    read:()=>{ const m=+$('#tcM').value,k=+$('#tcK').value;
      const A=1+m*m,B=2*m*k,C=k*k-4,D=B*B-4*A*C,tol=0.01;
      return [m.toFixed(1),k.toFixed(3),D.toFixed(4),D>tol?2:(D<-tol?0:1),
        (Math.abs(k)/Math.sqrt(1+m*m)).toFixed(3)]; } });
}
