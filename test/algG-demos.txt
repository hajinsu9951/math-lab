/* ---------- LAB 142 조합과 파스칼 삼각형 — 위 두 수의 합 ---------- */
function pascalDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="ps1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">각 자리는 <b>바로 위 두 수의 합</b>입니다.
      그 수가 곧 「n 개에서 r 개를 뽑는 방법의 수」입니다.</div>
    </div>
    <div class="col">
      ${slider('psN','줄 번호 n',1,12,1,6,'')}
      ${slider('psR','뽑는 개수 r',0,12,1,2,'')}
      <div class="ctl" style="margin-top:8px"><button class="b g sm" id="psSum">한 줄을 모두 더하기</button></div>
      <div class="out" id="psout"></div>
      <div class="note" style="margin-top:12px">C(n, r) = C(n−1, r−1) + C(n−1, r).

    </div></div>`;
  const cv=$('#ps1'),ctx=cv.getContext('2d'),W=380,H=320;
  const S={ sum:false };
  const C=(n,r)=>{ if(r<0||r>n) return 0; let v=1; for(let i=0;i<r;i++) v=v*(n-i)/(i+1); return Math.round(v); };
  function draw(){
    let n=+$('#psN').value, r=Math.min(+$('#psR').value,n);
    $('#psR').max=n; $('#psNv').textContent=n; $('#psRv').textContent=r;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const rows=n+1, dy=Math.min(26,(H-40)/rows), dx=Math.min(30,320/(n+1));
    for(let i=0;i<rows;i++){ for(let j=0;j<=i;j++){
      const px=W/2+(j-i/2)*dx, py=24+i*dy, v=C(i,j);
      const hit = i===n&&j===r, par = i===n-1&&(j===r-1||j===r);
      ctx.fillStyle = hit?'rgba(190,18,60,.85)':(par?'rgba(14,147,132,.75)':'#6B7280');
      ctx.font=(hit?'bold ':'')+Math.min(12,dx*0.44)+'px sans-serif'; ctx.textAlign='center';
      if(hit){ ctx.fillStyle='rgba(190,18,60,.14)'; ctx.beginPath(); ctx.arc(px,py-4,dx*0.44,0,7); ctx.fill();
        ctx.fillStyle='#BE123C'; }
      ctx.fillText(v, px, py); } }
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('붉은 = C(n, r) · 초록 = 그 위의 두 수', 14, H-10);
    const val=C(n,r), a=C(n-1,r-1), b=C(n-1,r), tot=Math.pow(2,n);
    $('#psout').innerHTML =
      `<span class="mono">C(${n}, ${r}) = <b style="color:#BE123C">${val}</b></span><br>`
      + `<span class="mono">위 두 수 ${a} + ${b} = <b>${a+b}</b> — 파스칼 항등식</span><br>`
      + (n===10&&r===3 ? `<b style="color:#B45309">오늘의 문제 — 10명에서 3명을 뽑는 방법은 ${val}가지입니다.</b><br>` : '')
      + (S.sum ? `<b style="color:#7C3AED">한 줄을 모두 더하면 ${tot} = 2^${n} — 부분집합의 개수와 같습니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">한 사람을 정해 「뽑는다 / 안 뽑는다」로 나누면 항등식이 보입니다.</span>`;
  }
  $('#psSum').onclick=()=>{ S.sum=true; draw(); };
  ['psN','psR'].forEach(i=>$('#'+i).oninput=()=>{ S.sum=false; draw(); }); draw();
  inquiry(root,{ id:'ps', title:'조합 — 파스칼 삼각형으로 세기',
    q:['(n, r, C(n,r), 위 두 수의 합) 을 여섯 줄 기록하세요.',
       '한 줄의 합이 2ⁿ 인 이유를 부분집합으로 설명하세요.',
       'C(n, r) = C(n, n−r) 인 이유를 「뽑는 대신 남기는」 관점으로 쓰세요.'],
    cols:['n','r','C(n,r)','위 두 수','합'],
    hint:'힌트 — 한 사람을 정해 뽑히는 경우와 아닌 경우로 나누면 항등식이 나옵니다.',
    read:()=>{ const n=+$('#psN').value,r=Math.min(+$('#psR').value,n);
      return [n,r,C(n,r),C(n-1,r-1)+' + '+C(n-1,r),C(n-1,r-1)+C(n-1,r)]; } });
}

    ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(0),Y(t)); ctx.lineTo(X(60-t),Y(60)); ctx.lineTo(X(60),Y(60));
    ctx.lineTo(X(60),Y(60-t)); ctx.lineTo(X(t),Y(0)); ctx.closePath(); ctx.fill();
    ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1; ctx.strokeRect(L,T,SZ,SZ);
    S.pts.forEach(p=>{ ctx.fillStyle = p[2]?'rgba(14,147,132,.75)':'rgba(190,18,60,.5)';
      ctx.beginPath(); ctx.arc(X(p[0]),Y(p[1]),1.7,0,7); ctx.fill(); });
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('내 도착 시각 →', L, T+SZ+18);
    ctx.save(); ctx.translate(16,T+SZ); ctx.rotate(-Math.PI/2); ctx.fillText('친구 도착 시각 →',0,0); ctx.restore();
    const p=1-Math.pow((60-t)/60,2);
    const sim = S.pts.length ? (S.hit/S.pts.length*100) : null;
    $('#mpout').innerHTML =
      `<span class="mono">이론값 = 1 − (${60-t}/60)² = <b style="color:#0E9384">${(p*100).toFixed(1)}%</b></span><br>`
      + (t===10 ? `<b style="color:#B45309">오늘의 문제 — 10분만 기다려 주면 만날 확률 ${(p*100).toFixed(1)}% 입니다.</b><br>` : '')
      + (sim!==null ? `<b style="color:#7C3AED">${S.pts.length}번 실험 결과 ${sim.toFixed(1)}% — 이론값과의 차이 ${Math.abs(sim-p*100).toFixed(1)}%p</b><br>` : '')
      + `<span class="mono">띠의 넓이 비 = ${(p).toFixed(4)}</span><br>`
      + `<span style="color:#6B7280;font-size:12.5px">만나지 못하는 두 삼각형의 넓이를 빼는 것이 더 쉽습니다.</span>`;
  }
  $('#mpRun').onclick=()=>{ const t=+$('#mpT').value; S.pts=[]; S.hit=0;
    for(let i=0;i<1000;i++){ const x=Math.random()*60, y=Math.random()*60, ok=Math.abs(x-y)<=t;
      if(ok) S.hit++; S.pts.push([x,y,ok]); } draw(); };
  $('#mpClr').onclick=()=>{ S.pts=[]; S.hit=0; draw(); };
  $('#mpT').oninput=()=>{ S.pts=[]; S.hit=0; draw(); }; draw();
  inquiry(root,{ id:'mp', title:'기하적 확률 — 넓이로 재는 확률',
    q:['t 를 바꾸며 (t, 이론값, 실험값) 을 다섯 줄 기록하세요.',
       '확률이 50% 를 넘으려면 몇 분을 기다려야 하나요?',
       '만나지 못하는 경우가 왜 두 개의 삼각형인지 그림으로 설명하세요.'],
    cols:['t','이론값(%)','실험 횟수','실험값(%)','차이'],
    hint:'힌트 — 만나지 못할 확률은 ((60−t)/60)² 입니다.',
    read:()=>{ const t=+$('#mpT').value, p=(1-Math.pow((60-t)/60,2))*100;
      const sim=S.pts.length?S.hit/S.pts.length*100:null;
      return [t,p.toFixed(1),S.pts.length,sim===null?'—':sim.toFixed(1),sim===null?'—':Math.abs(sim-p).toFixed(1)]; } });
