/* ---------- LAB 317 복소함수 z² — 각을 두 배로 ---------- */
function complexMapDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="cx1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">z² 은 <b>크기를 제곱하고 각을 두 배</b>로 합니다.
      크기만 제곱하면 격자가 휘지 않습니다.</div>
    </div>
    <div class="col">
      ${slider('cxT','점의 각',0,180,5,30,'°')}
      ${slider('cxR','점의 크기',0.4,1.6,0.05,1,'')}
      <label class="ctl" style="display:flex;align-items:center;gap:8px;margin-top:6px">
        <input type="checkbox" id="cxBad"> 각은 그대로 두고 크기만 제곱하기</label>
      <div class="out" id="cxout"></div>
    </div></div>`;
  const cv=$('#cx1'),ctx=cv.getContext('2d'),W=380,H=330;
  function draw(){
    const deg=+$('#cxT').value, r=+$('#cxR').value, bad=$('#cxBad').checked;
    $('#cxTv').textContent=deg+'°'; $('#cxRv').textContent=r.toFixed(2);
    const th=deg*Math.PI/180;
    const outT = bad ? th : 2*th, outR = r*r;
    const z=[r*Math.cos(th), r*Math.sin(th)];
    const w=[outR*Math.cos(outT), outR*Math.sin(outT)];
    const exact=[r*r*Math.cos(2*th), r*r*Math.sin(2*th)];
    const err=Math.hypot(w[0]-exact[0], w[1]-exact[1]);
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const cx=190,cy=190,u=78;
    const X=v=>cx+v*u, Y=v=>cy-v*u;
    ctx.strokeStyle='#F3F4F6'; ctx.lineWidth=1;
    for(let k=-2;k<=2;k++){ ctx.beginPath(); ctx.moveTo(X(k),14); ctx.lineTo(X(k),H-24); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(14,Y(k)); ctx.lineTo(W-14,Y(k)); ctx.stroke(); }
    ctx.strokeStyle='#D1D5DB'; ctx.lineWidth=1.2;
    ctx.beginPath(); ctx.moveTo(14,Y(0)); ctx.lineTo(W-14,Y(0)); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(X(0),14); ctx.lineTo(X(0),H-24); ctx.stroke();
    // 격자 한 줄이 어떻게 휘는가
    ctx.strokeStyle='rgba(37,99,235,.45)'; ctx.lineWidth=1.4;
    for(let k=1;k<=3;k++){ const rr=k*0.5;
      ctx.beginPath();
      for(let i=0;i<=90;i++){ const t=Math.PI*i/90;
        const T=bad?t:2*t, R=rr*rr;
        const px=X(R*Math.cos(T)), py=Y(R*Math.sin(T));
        i?ctx.lineTo(px,py):ctx.moveTo(px,py); }
      ctx.stroke(); }
    ctx.strokeStyle='#0E9384'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(z[0]),Y(z[1])); ctx.stroke();
    ctx.fillStyle='#0E9384'; ctx.beginPath(); ctx.arc(X(z[0]),Y(z[1]),5,0,7); ctx.fill();
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(w[0]),Y(w[1])); ctx.stroke();
    ctx.fillStyle='#BE123C'; ctx.beginPath(); ctx.arc(X(w[0]),Y(w[1]),5,0,7); ctx.fill();
    ctx.strokeStyle='rgba(190,18,60,.5)'; ctx.lineWidth=1.4;
    ctx.beginPath(); ctx.arc(X(0),Y(0),22,-th,-outT,outT>th); ctx.stroke();
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('초록 = z · 붉은 = 옮겨진 점 · 파랑 = 반원이 옮겨 간 자취', 14, 20);
    ctx.fillText('z = '+r.toFixed(2)+' ∠ '+deg+'°  →  '+outR.toFixed(3)+' ∠ '+(bad?deg:2*deg)+'°', 14, H-12);
    $('#cxout').innerHTML =
      `<span class="mono">z 의 크기 ${r.toFixed(2)} → 제곱 <b>${outR.toFixed(4)}</b> · 각 ${deg}° → <b style="color:${bad?'#BE123C':'#0E9384'}">${bad?deg:2*deg}°</b></span><br>`
      + `<span class="mono">정확한 z² = ${exact[0].toFixed(4)} + ${exact[1].toFixed(4)} i · 지금 값과의 차이 <b>${err.toFixed(4)}</b></span><br>`
      + (!bad && deg>=60 ? `<b style="color:#B45309">각이 두 배가 되어 반원이 원 전체를 덮습니다 — z² 은 평면을 두 겹으로 감습니다.</b><br>` : '')
      + (bad ? `<b style="color:#7C3AED">각을 그대로 두니 격자가 휘지 않습니다 — 크기만 제곱한 것은 z² 이 아닙니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">(r∠θ)² = r²∠2θ — 복소수의 곱은 크기는 곱하고 각은 더합니다.</span>`;
  }
  ['cxT','cxR'].forEach(i=>$('#'+i).oninput=draw); $('#cxBad').onchange=draw; draw();
  inquiry(root,{ id:'cx', title:'z² — 각을 두 배로',
    q:['각을 바꿔 가며 (각, 옮겨진 각) 을 여섯 줄 기록하세요.',
       '반원이 원 전체를 덮는 것을 확인하세요.',
       '크기만 제곱하면 무엇이 달라지는지 쓰세요.'],
    cols:['크기','각','옮겨진 크기','옮겨진 각'],
    hint:'힌트 — 복소수의 곱은 크기끼리 곱하고 각끼리 더합니다.',
    read:()=>{ const deg=+$('#cxT').value,r=+$('#cxR').value,bad=$('#cxBad').checked;
      return [r.toFixed(2),deg+'°',(r*r).toFixed(4),(bad?deg:2*deg)+'°']; } });
}

/* ---------- LAB 319 정삼각형의 대칭군 — 순서를 타는 곱 ---------- */

/* ---------- LAB 320 가정을 바꾸면 — 모형의 민감도 ---------- */

  root.innerHTML=`<div class="row"><div class="col">

      <div class="note" style="margin-top:8px">같은 자료라도 <b>어떤 가정을 두느냐</b>에 따라

    </div>
    <div class="col">
      <label class="ctl">성장 가정<select id="snK" style="width:100%;padding:7px 9px;border:1px solid #E5E7EB;border-radius:8px;font:inherit">
        <option value="exp">계속 같은 비율로 늘어난다 (지수)</option>
        <option value="log">한계가 있어 완만해진다 (로지스틱)</option>
        <option value="lin">일정한 양씩 늘어난다 (선형)</option>
      </select></label>
      ${slider('snR','성장률',5,40,1,20,'%')}
      ${slider('snT','예측할 기간',1,36,1,12,'개월')}
      <div class="out" id="snout"></div>
    </div></div>`;
  const cv=$('#sn1'),ctx=cv.getContext('2d'),W=380,H=330;
  const DATA=[100,118,140,166,192,225];               // 관측된 6 개월
  function model(k,r,t){
    const g=r/100;
    if(k==='exp') return 100*Math.pow(1+g,t);
    if(k==='lin') return 100+100*g*t;
    const K=900; return K/(1+(K/100-1)*Math.exp(-g*t));
  }
  function draw(){

    $('#snRv').textContent=r+'%'; $('#snTv').textContent=T+'개월';
    const fit=DATA.reduce((s,v,i)=>s+Math.pow(v-model(k,r,i),2),0)/DATA.length;
    const vals={exp:model('exp',r,T),log:model('log',r,T),lin:model('lin',r,T)};
    const spread=Math.max(...Object.values(vals))/Math.max(1,Math.min(...Object.values(vals)));
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const x0=40,y0=260,gw=W-64,gh=210;
    const mx=Math.max(...Object.values(vals),1000);
    const X=t=>x0+gw*t/36, Y=v=>y0-gh*Math.min(v,mx)/mx;
    ctx.strokeStyle='#D1D5DB'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(x0,y0); ctx.lineTo(x0+gw,y0); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x0,y0-gh); ctx.lineTo(x0,y0); ctx.stroke();
    ctx.fillStyle='rgba(37,99,235,.06)'; ctx.fillRect(x0,y0-gh,X(5)-x0,gh);
    const COL={exp:'#BE123C',log:'#0E9384',lin:'#B45309'};
    ['exp','log','lin'].forEach(m=>{
      ctx.strokeStyle= m===k ? COL[m] : 'rgba(156,163,175,.55)';
      ctx.lineWidth= m===k ? 2.6 : 1.3;
      ctx.beginPath();
      for(let i=0;i<=180;i++){ const t=36*i/180;
        i?ctx.lineTo(X(t),Y(model(m,r,t))):ctx.moveTo(X(t),Y(model(m,r,t))); }
      ctx.stroke(); });
    DATA.forEach((v,i)=>{ ctx.fillStyle='#111827';
      ctx.beginPath(); ctx.arc(X(i),Y(v),4,0,7); ctx.fill(); });
    ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1.2; ctx.setLineDash([4,3]);
    ctx.beginPath(); ctx.moveTo(X(T),y0-gh); ctx.lineTo(X(T),y0); ctx.stroke(); ctx.setLineDash([]);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('검은 점 = 관측한 6 개월 (옅은 파랑 구간) · 진한 선 = 고른 가정', 14, 20);
    ctx.fillText('세 가정의 '+T+'개월 예측 : 지수 '+Math.round(vals.exp)
      +' · 로지스틱 '+Math.round(vals.log)+' · 선형 '+Math.round(vals.lin), 14, H-12);
    $('#snout').innerHTML =
      `<span class="mono">관측 구간의 맞춤 오차 = <b>${fit.toFixed(1)}</b> · 세 가정 모두 초반은 비슷하게 맞습니다</span><br>`
      + `<span class="mono">${T} 개월 예측의 최대/최소 배율 = <b style="color:${spread>=3?'#BE123C':'#0E9384'}">${spread.toFixed(2)}</b> 배</span><br>`
      + (T>=24 && spread>=3 ? `<b style="color:#B45309">같은 자료에서 출발했는데 예측이 ${spread.toFixed(1)} 배나 벌어졌습니다 — 가정이 결론을 정한 것입니다.</b><br>` : '')
      + (k==='log' && T>=24 ? `<b style="color:#7C3AED">한계를 가정하면 곡선이 눕습니다 — 어떤 가정을 두었는지 반드시 밝혀야 합니다.</b><br>` : '')

  }
  $('#snK').onchange=draw; ['snR','snT'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'sn', title:'민감도 — 가정이 결론을 정한다',

       '세 예측이 벌어지기 시작하는 시점을 찾으세요.',

}

/* ---------- LAB 321 외삽의 위험 — 어디까지 믿어도 될까 ---------- */
function extrapolateDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="ex1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">자료가 있는 구간 안에서 맞히는 것과
      <b>밖으로 늘여 예측하는 것</b>은 전혀 다른 일입니다.</div>
    </div>
    <div class="col">
      ${slider('exD','맞춤 곡선의 차수',1,6,1,1,'차')}
      ${slider('exX','예측할 지점 x',0,16,0.5,5,'')}
      <div class="out" id="exout"></div>
    </div></div>`;
  const cv=$('#ex1'),ctx=cv.getContext('2d'),W=380,H=330;
  const XS=[1,2,3,4,5,6,7,8], YS=[2.1,3.9,6.2,7.8,10.1,12.2,13.8,16.1];
  function fit(deg){
    const n=deg+1;
    const A=[], B=[];
    for(let i=0;i<n;i++){ A.push([]);
      for(let j=0;j<n;j++) A[i].push(XS.reduce((s,x)=>s+Math.pow(x,i+j),0));
      B.push(XS.reduce((s,x,k)=>s+YS[k]*Math.pow(x,i),0)); }
    for(let i=0;i<n;i++){
      let p=i; for(let r=i+1;r<n;r++) if(Math.abs(A[r][i])>Math.abs(A[p][i])) p=r;
      [A[i],A[p]]=[A[p],A[i]]; [B[i],B[p]]=[B[p],B[i]];
      if(Math.abs(A[i][i])<1e-12) continue;
      for(let r=0;r<n;r++){ if(r===i) continue;
        const f=A[r][i]/A[i][i];
        for(let c=i;c<n;c++) A[r][c]-=f*A[i][c];
        B[r]-=f*B[i]; } }
    return A.map((row,i)=>Math.abs(row[i])<1e-12?0:B[i]/row[i]);
  }
  const evalp=(c,x)=>c.reduce((s,a,i)=>s+a*Math.pow(x,i),0);
  function draw(){
    const d=+$('#exD').value, px=+$('#exX').value;
    $('#exDv').textContent=d+'차'; $('#exXv').textContent=px.toFixed(1);
    const c=fit(d);
    const inRange = px>=1 && px<=8;
    const pred=evalp(c,px);
    const rmse=Math.sqrt(XS.reduce((s,x,i)=>s+Math.pow(YS[i]-evalp(c,x),2),0)/XS.length);
    const linear=2*px;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const x0=34,y0=270,u=20.5,v=13;
    const X=t=>x0+t*u, Y=w=>y0-Math.max(-4,Math.min(20,w))*v;
    ctx.fillStyle='rgba(37,99,235,.07)'; ctx.fillRect(X(1),30,X(8)-X(1),y0-30);
    ctx.strokeStyle='#D1D5DB'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(x0,y0); ctx.lineTo(X(17),y0); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(x0,30); ctx.lineTo(x0,y0); ctx.stroke();
    ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1.2; ctx.setLineDash([4,3]);
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(17),Y(34)); ctx.stroke(); ctx.setLineDash([]);
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=2.4; ctx.beginPath();
    let st=false;
    for(let i=0;i<=340;i++){ const t=17*i/340; const w=evalp(c,t);
      if(w<-4||w>20){ st=false; continue; }
      st?ctx.lineTo(X(t),Y(w)):ctx.moveTo(X(t),Y(w)); st=true; }
    ctx.stroke();
    XS.forEach((x,i)=>{ ctx.fillStyle='#111827';
      ctx.beginPath(); ctx.arc(X(x),Y(YS[i]),4,0,7); ctx.fill(); });
    ctx.fillStyle=inRange?'#0E9384':'#BE123C';
    ctx.beginPath(); ctx.arc(X(px),Y(Math.max(-4,Math.min(20,pred))),6,0,7); ctx.fill();
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('옅은 파랑 = 자료가 있는 구간 · 회색 점선 = 눈대중 직선', 14, 20);
    ctx.fillText('맞춤 오차(RMSE) = '+rmse.toFixed(4), 14, H-12);
    $('#exout').innerHTML =
      `<span class="mono">${d} 차 곡선 · 맞춤 오차 = <b style="color:#0E9384">${rmse.toFixed(4)}</b> — 차수를 올릴수록 자료에는 더 잘 맞습니다</span><br>`
      + `<span class="mono">x = ${px.toFixed(1)} 예측 = <b style="color:${inRange?'#0E9384':'#BE123C'}">${pred.toFixed(2)}</b> · 눈대중 직선이라면 ${linear.toFixed(1)}</span><br>`
      + (!inRange && Math.abs(pred-linear)>=6 ? `<b style="color:#B45309">자료 구간 밖으로 나가자 예측이 눈대중과 ${Math.abs(pred-linear).toFixed(0)} 이나 벌어졌습니다 — 외삽은 위험합니다.</b><br>` : '')
      + (d>=5 ? `<b style="color:#7C3AED">차수를 높이면 자료에는 더 잘 맞는데 구간 밖에서는 더 크게 튑니다 — 과적합입니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">맞춤이 좋다고 예측이 좋은 것은 아닙니다 — 적용 범위를 반드시 밝혀야 합니다.</span>`;
  }
  ['exD','exX'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'ex', title:'외삽 — 어디까지 믿어도 될까',
    q:['차수와 예측 지점을 바꿔 가며 (차수, 맞춤 오차, 예측값) 을 여섯 줄 기록하세요.',
       '자료 구간 안과 밖에서 예측이 어떻게 다른지 쓰세요.',
       '차수를 높이면 무엇이 좋아지고 무엇이 나빠지는지 정리하세요.'],
    cols:['차수','예측 지점','예측값','맞춤 오차','구간 안팎'],
    hint:'힌트 — 맞춤 오차가 작아진다고 예측이 좋아지지는 않습니다.',
    read:()=>{ const d=+$('#exD').value,px=+$('#exX').value,c=fit(d);
      const rmse=Math.sqrt(XS.reduce((s,x,i)=>s+Math.pow(YS[i]-evalp(c,x),2),0)/XS.length);
      return [d+'차',px.toFixed(1),evalp(c,px).toFixed(2),rmse.toFixed(4),(px>=1&&px<=8)?'안':'밖']; } });
}
