/* ---------- LAB 167 삼수선 정리 — 공간에서 수직을 확인하는 법 ---------- */
function threePerpDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="tp1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">평면 위의 직선 ℓ, 평면 밖의 점 P, 발 H, 그리고 ℓ 위의 점 A.
      <b>세 개의 수직</b>이 서로 어떻게 이어지는지 보세요.</div>
    </div>
    <div class="col">
      ${slider('tpH','P 의 높이 (PH)',1,6,0.5,3,'')}
      ${slider('tpD','H 에서 ℓ 까지의 거리 (HA)',0,5,0.5,0,'')}
      ${slider('tpS','A 를 ℓ 위에서 옮기기',-4,4,0.5,0,'')}
      <div class="out" id="tpout"></div>
      <div class="note" style="margin-top:12px">PH ⊥ 평면, HA ⊥ ℓ 이면 <b>PA ⊥ ℓ</b>.
      셋 중 둘이 수직이면 나머지 하나도 수직이 됩니다.</div>
    </div></div>`;
  const cv=$('#tp1'),ctx=cv.getContext('2d'),W=380,H=330;
  function draw(){
    const h=+$('#tpH').value, d=+$('#tpD').value, sv=+$('#tpS').value;
    $('#tpHv').textContent=h.toFixed(1); $('#tpDv').textContent=d.toFixed(1); $('#tpSv').textContent=sv.toFixed(1);
    // 비스듬히 본 평면: x 는 오른쪽, y 는 안쪽(비스듬), z 는 위
    const pr=(x,y,z)=>[70+30*x+16*y, 240-30*z+12*y];
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='rgba(37,99,235,.07)'; ctx.beginPath();
    [[-1.5,-1.5],[7,-1.5],[7,4],[-1.5,4]].forEach((c,i)=>{ const q=pr(c[0],c[1],0); i?ctx.lineTo(q[0],q[1]):ctx.moveTo(q[0],q[1]); });
    ctx.closePath(); ctx.fill(); ctx.strokeStyle='#C7D2E4'; ctx.lineWidth=1; ctx.stroke();
    const Hh=[0,0,0], A=[0,d,0], P=[0,0,h];
    const L0=pr(-1.2,d,0), L1=pr(6.6,d,0);
    ctx.strokeStyle='#0E9384'; ctx.lineWidth=2.4;
    ctx.beginPath(); ctx.moveTo(L0[0],L0[1]); ctx.lineTo(L1[0],L1[1]); ctx.stroke();
    const Ap=pr(sv,d,0), Hp=pr(0,0,0), Pp=pr(0,0,h);
    const seg=(a,b,col,w,dash)=>{ ctx.strokeStyle=col; ctx.lineWidth=w;
      if(dash) ctx.setLineDash([4,4]);
      ctx.beginPath(); ctx.moveTo(a[0],a[1]); ctx.lineTo(b[0],b[1]); ctx.stroke(); ctx.setLineDash([]); };
    seg(Hp,Pp,'#BE123C',2.6);
    seg(Hp,Ap,'#2563EB',2.2);
    seg(Pp,Ap,'#7C3AED',2.6);
    [[Pp,'P','#BE123C'],[Hp,'H','#111827'],[Ap,'A','#2563EB']].forEach(([q,nm,col])=>{
      ctx.fillStyle=col; ctx.beginPath(); ctx.arc(q[0],q[1],4.6,0,7); ctx.fill();
      ctx.fillStyle='#374151'; ctx.font='12px sans-serif'; ctx.textAlign='left';
      ctx.fillText(nm, q[0]+7, q[1]-6); });
    ctx.fillStyle='#0E9384'; ctx.font='10.5px sans-serif';
    ctx.fillText('직선 ℓ', L1[0]-40, L1[1]-8);
    ctx.fillStyle='#6B7280';
    ctx.fillText('붉은 PH ⊥ 평면 · 파랑 HA · 보라 PA', 12, H-14);
    // 벡터 계산 (실제 3차원 좌표)
    const PA=[sv-0, d-0, -h], l=[1,0,0];
    const dotPA = PA[0]*l[0]+PA[1]*l[1]+PA[2]*l[2];
    const HA=[sv-0, d-0, 0], dotHA = HA[0];
    const lenPA=Math.hypot(PA[0],PA[1],PA[2]);
    $('#tpout').innerHTML =
      `<span class="mono">PH = ${h.toFixed(1)} · HA = ${Math.hypot(sv,d).toFixed(2)} · PA = <b>${lenPA.toFixed(3)}</b></span><br>`
      + `<span class="mono">HA·ℓ = <b>${dotHA.toFixed(2)}</b> · PA·ℓ = <b>${dotPA.toFixed(2)}</b></span><br>`
      + (Math.abs(dotHA)<1e-9 && Math.abs(dotPA)<1e-9 && d>0.05
          ? `<b style="color:#B45309">HA 가 ℓ 에 수직이니 PA 도 ℓ 에 수직입니다 — 삼수선 정리 그대로입니다.</b><br>` : '')
      + (Math.abs(dotHA)>1e-9
          ? `<b style="color:#7C3AED">A 를 옮기자 HA 가 ℓ 에 수직이 아니게 되었고, PA 도 수직이 아닙니다.</b><br>` : '')
      + (d<0.05 ? `H 가 ℓ 위에 있어 HA 가 사라졌습니다 — 정리를 쓸 수 없는 경우입니다.<br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">PA² = PH² + HA² 도 함께 확인해 보세요.</span>`;
  }
  ['tpH','tpD','tpS'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'tp', title:'삼수선 정리 — 수직이 옮겨 간다',
    q:['A 를 옮기며 (HA·ℓ, PA·ℓ) 를 여섯 줄 기록하세요 — 두 값이 함께 0 이 되나요?',
       'PA² = PH² + HA² 가 성립하는지 값으로 확인하세요.',
       '「PH ⊥ 평면, HA ⊥ ℓ ⟹ PA ⊥ ℓ」을 자기 말로 정리하세요.'],
    cols:['PH','HA','PA','HA·ℓ','PA·ℓ'],
    hint:'힌트 — A 가 H 의 바로 앞(수직 발)일 때만 두 내적이 함께 0 이 됩니다.',
    read:()=>{ const h=+$('#tpH').value,d=+$('#tpD').value,sv=+$('#tpS').value;
      return [h.toFixed(1),Math.hypot(sv,d).toFixed(2),Math.hypot(sv,d,h).toFixed(3),
        sv.toFixed(2),sv.toFixed(2)]; } });
}

/* ---------- LAB 169 구를 자르면 — 구의 방정식과 평면의 절단 ---------- */
function sphereCutDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="sc1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">구를 수평면으로 자릅니다.
      자른 높이에 따라 <b>단면의 크기</b>가 어떻게 변하는지 보세요.</div>
    </div>
    <div class="col">
      ${slider('scR','구의 반지름 R',2,6,0.5,4,'')}
      ${slider('scH','자르는 높이 h',-6,6,0.1,1.5,'')}
      <div class="out" id="scout"></div>
      <div class="note" style="margin-top:12px">x² + y² + z² = R² 에 z = h 를 넣으면
      x² + y² = R² − h² — <b>반지름 √(R² − h²) 인 원</b>입니다.</div>
    </div></div>`;
  const cv=$('#sc1'),ctx=cv.getContext('2d'),W=380,H=330;
  function draw(){
    const R=+$('#scR').value; let h=+$('#scH').value;
    $('#scRv').textContent=R.toFixed(1); $('#scHv').textContent=h.toFixed(1);
    const sc=26, cx=190, cy=150;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='rgba(37,99,235,.10)';
    ctx.beginPath(); ctx.arc(cx,cy,R*sc,0,7); ctx.fill();
    ctx.strokeStyle='#2563EB'; ctx.lineWidth=2.2; ctx.stroke();
    ctx.strokeStyle='rgba(37,99,235,.35)'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.ellipse(cx,cy,R*sc,R*sc*0.32,0,0,7); ctx.stroke();
    const inside=Math.abs(h)<R;
    const r = inside ? Math.sqrt(R*R-h*h) : 0;
    const y = cy-h*sc;
    ctx.strokeStyle=inside?'#BE123C':'#9CA3AF'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(24,y); ctx.lineTo(W-24,y); ctx.stroke();
    if(inside){ ctx.fillStyle='rgba(190,18,60,.25)';
      ctx.beginPath(); ctx.ellipse(cx,y,r*sc,r*sc*0.32,0,0,7); ctx.fill();
      ctx.strokeStyle='#BE123C'; ctx.lineWidth=1.8; ctx.stroke(); }
    ctx.fillStyle='#374151'; ctx.font='11px sans-serif'; ctx.textAlign='left';
    ctx.fillText('z = '+h.toFixed(1), W-70, y-6);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif';
    ctx.fillText('붉은 타원 = 잘린 단면 (실제로는 원)', 16, H-14);
    $('#scout').innerHTML =
      `<span class="mono">x² + y² + z² = ${(R*R).toFixed(2)} 에 z = ${h.toFixed(1)} 을 대입</span><br>`
      + (inside ? `<span class="mono">x² + y² = ${(R*R-h*h).toFixed(2)} → 반지름 <b style="color:#BE123C">${r.toFixed(3)}</b> 인 원 · 넓이 ${(Math.PI*r*r).toFixed(2)}</span><br>`
                : `<span class="mono">R² − h² = ${(R*R-h*h).toFixed(2)} < 0 — 해가 없습니다.</b></span><br>`)
      + (inside && Math.abs(h)<0.05 ? `<b style="color:#B45309">중심을 지나게 잘랐습니다 — 단면이 가장 큰 원(대원)입니다.</b><br>` : '')
      + (!inside ? `<b style="color:#7C3AED">평면이 구를 벗어나 만나지 않습니다 — 단면이 생기지 않습니다.</b><br>` : '')
      + (inside && Math.abs(Math.abs(h)-R)<0.06 ? `평면이 구에 접합니다 — 단면이 점 하나로 줄었습니다.<br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">단면의 반지름은 h 에 비례하지 않습니다 — √(R² − h²) 입니다.</span>`;
  }
  ['scR','scH'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'sc', title:'구의 절단 — 언제나 원',
    q:['h 를 바꾸며 (h, 단면 반지름, 단면 넓이) 를 여섯 줄 기록하세요.',
       '단면이 가장 큰 h 와 단면이 사라지는 h 를 각각 찾으세요.',
       '단면 반지름이 h 에 비례하지 않음을 표로 보이세요.'],
    cols:['R','h','R²−h²','단면 반지름','넓이'],
    hint:'힌트 — z = h 를 대입하면 x² + y² = R² − h² 입니다.',
    read:()=>{ const R=+$('#scR').value,h=+$('#scH').value,k=R*R-h*h;
      return [R.toFixed(1),h.toFixed(1),k.toFixed(2),k>0?Math.sqrt(k).toFixed(3):'없음',
        k>0?(Math.PI*k).toFixed(2):'0']; } });
}

/* ---------- LAB 170 원뿔을 어떻게 자르느냐 — 이차곡선의 통일 ---------- */
function coneCutDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="cc1" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">같은 원뿔을 <b>자르는 각도만</b> 바꿉니다.
      원 · 타원 · 포물선 · 쌍곡선이 차례로 나타납니다.</div>
    </div>
    <div class="col">
      ${slider('ccT','자르는 평면의 기울기',0,80,5,20,'°')}
      <div class="out" id="ccout"></div>
      <div class="note" style="margin-top:12px">원뿔의 모선 각을 45°라 하면,
      자르는 각이 <b>45°보다 작으면 타원, 같으면 포물선, 크면 쌍곡선</b>입니다.
      이심률 e = tan(자른 각)/tan(모선 각).</div>
    </div></div>`;
  const cv=$('#cc1'),ctx=cv.getContext('2d'),W=380,H=330;
  const CONE=45;
  function draw(){
    const t=+$('#ccT').value; $('#ccTv').textContent=t+'°';
    const e=Math.tan(t*Math.PI/180)/Math.tan(CONE*Math.PI/180);
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const apex=[110,40], sc=1;
    ctx.strokeStyle='#2563EB'; ctx.lineWidth=2;
    ctx.beginPath(); ctx.moveTo(apex[0],apex[1]); ctx.lineTo(apex[0]-80,apex[1]+160); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(apex[0],apex[1]); ctx.lineTo(apex[0]+80,apex[1]+160); ctx.stroke();
    ctx.strokeStyle='rgba(37,99,235,.5)'; ctx.lineWidth=1.4;
    ctx.beginPath(); ctx.ellipse(apex[0],apex[1]+160,80,22,0,0,7); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(apex[0],apex[1]); ctx.lineTo(apex[0]-80,apex[1]-160); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(apex[0],apex[1]); ctx.lineTo(apex[0]+80,apex[1]-160); ctx.stroke();
    const rad=t*Math.PI/180, cxp=apex[0], cyp=apex[1]+100;
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=2.4;
    ctx.beginPath();
    ctx.moveTo(cxp-110*Math.cos(rad), cyp+110*Math.sin(rad));
    ctx.lineTo(cxp+110*Math.cos(rad), cyp-110*Math.sin(rad));
    ctx.stroke();
    let kind, note;
    if(t===0){ kind='원'; note='자르는 면이 밑면과 나란합니다.'; }
    else if(t<CONE-0.5){ kind='타원'; note='한 겹만 잘라 닫힌 곡선이 됩니다.'; }
    else if(Math.abs(t-CONE)<0.5){ kind='포물선'; note='모선과 나란해 한쪽으로 열립니다.'; }
    else { kind='쌍곡선'; note='두 겹을 모두 잘라 두 갈래가 됩니다.'; }
    const bx=230, by=90, R=64;
    ctx.strokeStyle='#0E9384'; ctx.lineWidth=2.4;
    if(kind==='원'){ ctx.beginPath(); ctx.arc(bx+40,by+80,40,0,7); ctx.stroke(); }
    else if(kind==='타원'){ ctx.beginPath(); ctx.ellipse(bx+40,by+80,46,46*Math.sqrt(Math.max(0.02,1-e*e)),0,0,7); ctx.stroke(); }
    else if(kind==='포물선'){ ctx.beginPath();
      for(let i=0;i<=60;i++){ const u=-1.4+2.8*i/60, X=bx+40+38*u, Y=by+30+34*u*u;
        i?ctx.lineTo(X,Y):ctx.moveTo(X,Y); } ctx.stroke(); }
    else { [1,-1].forEach(s=>{ ctx.beginPath();
      for(let i=0;i<=60;i++){ const u=-1.3+2.6*i/60, X=bx+40+s*30*Math.cosh(u), Y=by+80+34*Math.sinh(u);
        i?ctx.lineTo(X,Y):ctx.moveTo(X,Y); } ctx.stroke(); }); }
    ctx.fillStyle='#374151'; ctx.font='bold 12px sans-serif'; ctx.textAlign='left';
    ctx.fillText('단면: '+kind, bx, by-6);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif';
    ctx.fillText('붉은 선 = 자르는 평면 · 모선 각 45°', 12, H-14);
    $('#ccout').innerHTML =
      `<span class="mono">자르는 각 ${t}° · 모선 각 ${CONE}° · 이심률 e = tan${t}°/tan45° = <b>${e.toFixed(3)}</b></span><br>`
      + `<span class="mono">단면의 이름 — <b style="color:#0E9384">${kind}</b> (${note})</span><br>`
      + (Math.abs(t-CONE)<0.5 ? `<b style="color:#B45309">모선과 나란해지는 순간 — 닫힌 곡선이 열려 포물선이 됩니다 (e = 1).</b><br>` : '')
      + (t>CONE+0.5 ? `<b style="color:#7C3AED">더 세우니 두 겹을 모두 잘라 두 갈래가 생겼습니다 — 쌍곡선입니다 (e > 1).</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">세 곡선은 서로 다른 도형이 아니라 같은 원뿔의 다른 단면입니다.</span>`;
  }
  $('#ccT').oninput=draw; draw();
  inquiry(root,{ id:'cc', title:'원뿔곡선 — 하나의 원뿔, 네 가지 단면',
    q:['각을 0°부터 80°까지 바꾸며 (각, e, 단면 이름) 을 여섯 줄 기록하세요.',
       '타원에서 포물선으로 바뀌는 경계의 각과 e 를 찾으세요.',
       '이심률이 0, 0.5, 1, 1.5 일 때의 도형을 각각 적으세요.'],
    cols:['자른 각','e','단면','열림/닫힘','겹 수'],
    hint:'힌트 — e = 1 이 되는 순간이 경계입니다.',
    read:()=>{ const t=+$('#ccT').value, e=Math.tan(t*Math.PI/180)/Math.tan(45*Math.PI/180);
      const kind = t===0?'원':(t<44.5?'타원':(Math.abs(t-45)<0.5?'포물선':'쌍곡선'));
      return [t+'°',e.toFixed(3),kind,kind==='원'||kind==='타원'?'닫힘':'열림',kind==='쌍곡선'?2:1]; } });
}

function posVecDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="pv1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">두 드론 A, B 사이를 <b>m : n 으로 나누는 자리</b>에

      <div class="note" style="margin-top:12px">내분점 P = (n·A + m·B)/(m + n).
      비가 <b>엇갈려 붙는다</b>는 점을 눈으로 확인하세요. 무게중심은 (A+B+C)/3.</div>

    ctx.beginPath(); ctx.moveTo(X(A[0]),Y(A[1])); ctx.lineTo(X(B[0]),Y(B[1])); ctx.stroke();
    let P=null;
    if(m+n>0){ P=[(n*A[0]+m*B[0])/(m+n), (n*A[1]+m*B[1])/(m+n)];
      ctx.fillStyle='#BE123C'; ctx.beginPath(); ctx.arc(X(P[0]),Y(P[1]),6,0,7); ctx.fill(); }

      (P ? `<span class="mono">P = (${n}·A + ${m}·B)/(${m}+${n}) = <b style="color:#BE123C">(${P[0].toFixed(2)}, ${P[1].toFixed(2)})</b></span><br>`
