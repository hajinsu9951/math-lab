/* ---------- LAB 307 귀류법 — 아니라고 가정하면 무엇이 무너지나 ---------- */
function reductioDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="rd2" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">「√2 가 유리수다」라고 <b>가정</b>한 뒤,
      그 가정에서 모순이 나오는지 따라갑니다.</div>
    </div>
    <div class="col">
      ${slider('rdS','증명의 단계',0,5,1,0,'단계')}
      <label class="ctl" style="display:flex;align-items:center;gap:8px;margin-top:6px">

      <div class="out" id="rdout"></div>
    </div></div>`;
  const cv=$('#rd2'),ctx=cv.getContext('2d'),W=380,H=330;
  const STEP=[
    ['가정','√2 = p / q (p, q 는 서로소인 자연수)'],
    ['양변 제곱','2 = p² / q²  →  p² = 2q²'],
    ['p 는 짝수','p² 이 짝수이므로 p 도 짝수 : p = 2k'],
    ['대입','(2k)² = 2q²  →  4k² = 2q²  →  q² = 2k²'],
    ['q 도 짝수','q² 이 짝수이므로 q 도 짝수'],
    ['모순','p 와 q 가 둘 다 짝수 — 서로소라던 가정과 어긋난다']
  ];
  const BAD=[
    ['잘못된 가정','√2 는 무리수다 (증명하려던 것을 그대로 가정)'],
    ['전개','그러므로 √2 = p / q 로 쓸 수 없다'],
    ['결론','따라서 √2 는 무리수다'],
    ['다시 보기','쓴 것이 곧 결론 — 아무것도 보이지 않았다'],
    ['','' ],['','']
  ];
  function draw(){
    const k=+$('#rdS').value, bad=$('#rdBad').checked;
    $('#rdSv').textContent=k+'단계';
    const L=bad?BAD:STEP;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText(bad?'가정을 부정하지 않은 「증명」':'귀류법 — 가정에서 모순으로', 16, 20);
    L.forEach(([ttl,body],i)=>{
      if(!ttl) return;
      const y=42+i*46, on=i<=k;
      ctx.fillStyle= on ? (i===5&&!bad ? 'rgba(190,18,60,.14)' : 'rgba(37,99,235,.10)') : '#F9FAFB';
      ctx.fillRect(16,y,W-32,40);
      ctx.strokeStyle= on ? (i===5&&!bad?'#BE123C':'#93C5FD') : '#F3F4F6';
      ctx.strokeRect(16,y,W-32,40);
      ctx.fillStyle= on ? '#111827' : '#D1D5DB';
      ctx.font='bold 11px sans-serif';
      ctx.fillText((i+1)+'. '+ttl, 26, y+16);
      ctx.font='11.5px sans-serif';
      ctx.fillStyle= on ? '#374151' : '#E5E7EB';
      ctx.fillText(body, 26, y+33); });
    const done = !bad && k>=5;
    $('#rdout').innerHTML =
      `<span class="mono">지금 단계 = ${k+1} / ${bad?4:6}</span><br>`
      + `<span class="mono">가정 : ${bad?'√2 는 무리수다 (증명할 것을 그대로 가정)':'√2 = p/q 로 쓸 수 있다 (증명할 것의 부정)'}</span><br>`
      + (done ? `<b style="color:#B45309">서로소라고 해 놓고 둘 다 짝수가 나왔습니다 — 가정이 무너졌습니다.</b><br>` : '')
      + (bad ? `<b style="color:#7C3AED">증명하려던 것을 그대로 가정했더니 결론이 가정의 되풀이가 되었습니다 — 순환논법입니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">귀류법은 「결론의 부정」에서 출발해야 합니다.</span>`;
  }
  $('#rdS').oninput=draw; $('#rdBad').onchange=draw; draw();
  inquiry(root,{ id:'rd2', title:'귀류법 — 부정에서 출발하기',
    q:['단계를 하나씩 올리며 (단계, 무엇이 밝혀졌는지) 를 여섯 줄 기록하세요.',
       '어느 단계에서 모순이 드러나는지 찾으세요.',
       '가정을 부정하지 않으면 왜 안 되는지 쓰세요.'],
    cols:['단계','내용','모순인가'],
    hint:'힌트 — 귀류법의 첫 줄은 언제나 「결론이 아니라고 하자」입니다.',
    read:()=>{ const k=+$('#rdS').value,bad=$('#rdBad').checked;
      const L=bad?BAD:STEP;
      return [k+1,(L[k]||['',''])[0],(!bad&&k>=5)?'모순':'아직']; } });
}

/* ---------- LAB 309 유리수 줄 세우기 — 분수도 자연수만큼 ---------- */
function countableDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="ct2" width="380" height="330"></canvas>
      <div class="note" style="margin-top:8px">분수를 <b>대각선으로</b> 훑으면 모든 분수에 번호가 붙습니다.
      분모별로 차례대로 세면 첫 줄에서 영영 끝나지 않습니다.</div>
    </div>
    <div class="col">
      ${slider('ctN','붙인 번호',1,60,1,1,'번')}
      <label class="ctl" style="display:flex;align-items:center;gap:8px;margin-top:6px">
        <input type="checkbox" id="ctRow"> 분모별로 차례대로 세기</label>
      <div class="out" id="ct2out"></div>
    </div></div>`;
  const cv=$('#ct2'),ctx=cv.getContext('2d'),W=380,H=330;
  const gcd=(a,b)=>b?gcd(b,a%b):a;
  function order(n,row){
    const out=[];
    if(row){ for(let q=1;q<=1&&out.length<n;q++) for(let p=1;out.length<n;p++) out.push([p,q]); return out; }
    for(let s=2;out.length<n;s++)
      for(let p=1;p<s && out.length<n;p++){ const q=s-p; out.push([p,q]); }
    return out;
  }
  function draw(){
    const n=+$('#ctN').value, row=$('#ctRow').checked;
    $('#ctNv').textContent=n+'번';
    const seq=order(n,row);
    const last=seq[seq.length-1];
    const seen=new Set(seq.map(([p,q])=>(p/gcd(p,q))+'/'+(q/gcd(p,q))));
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    const N=7, cw=46, ch=34, x0=54, y0=54;
    ctx.fillStyle='#6B7280'; ctx.font='10px sans-serif'; ctx.textAlign='center';
    for(let q=1;q<=N;q++) ctx.fillText('/'+q, 30, y0+(q-1)*ch+22);
    for(let p=1;p<=N;p++) ctx.fillText(p, x0+(p-1)*cw+cw/2, y0-8);
    for(let q=1;q<=N;q++) for(let p=1;p<=N;p++){
      const idx=seq.findIndex(([a,b])=>a===p&&b===q);
      const X=x0+(p-1)*cw, Y=y0+(q-1)*ch;
      ctx.fillStyle= idx>=0 ? 'rgba(37,99,235,'+(0.14+0.5*(1-idx/Math.max(n,1))).toFixed(2)+')' : '#F9FAFB';
      ctx.fillRect(X,Y,cw-4,ch-4);
      ctx.strokeStyle='#F3F4F6'; ctx.strokeRect(X,Y,cw-4,ch-4);
      if(idx>=0){ ctx.fillStyle='#1E3A8A'; ctx.font='bold 11px sans-serif'; ctx.textAlign='center';
        ctx.fillText(idx+1, X+(cw-4)/2, Y+21); } }
    ctx.textAlign='left'; ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif';
    ctx.fillText('칸 안의 수 = 그 분수에 붙은 번호 (가로 = 분자, 세로 = 분모)', 16, 22);
    ctx.fillText(row?'분모 1 인 줄만 세는 중':'대각선으로 훑는 중', 16, H-14);
    const reach = row ? (last?last[1]:1) : (last?last[0]+last[1]:2);
    $('#ct2out').innerHTML =
      `<span class="mono">${n} 번까지 붙였습니다 · 마지막 = <b style="color:#2563EB">${last?last[0]+'/'+last[1]:'—'}</b></span><br>`
      + `<span class="mono">서로 다른 분수 = ${seen.size} 개 · 지금 세고 있는 줄(분자+분모) = ${reach}</span><br>`
      + (!row && n>=30 ? `<b style="color:#B45309">번호를 붙이는 방법이 있으니 분수도 자연수만큼입니다 — 가산집합입니다.</b><br>` : '')
      + (row ? `<b style="color:#7C3AED">분모 1 인 줄만 세고 있어 다음 줄로 영영 넘어가지 못합니다 — 이 방법으로는 번호가 다 붙지 않습니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">「무한이 같은 크기다」는 곧 「일대일 대응이 있다」는 뜻입니다.</span>`;
  }
  $('#ctN').oninput=draw; $('#ctRow').onchange=draw; draw();
  inquiry(root,{ id:'ct2', title:'가산집합 — 번호를 붙일 수 있는가',
    q:['번호를 늘려 가며 (번호, 마지막 분수) 를 여섯 줄 기록하세요.',
       '분모별로 세면 왜 안 되는지 쓰세요.',
       '「크기가 같다」의 뜻을 일대일 대응으로 설명하세요.'],
    cols:['번호','마지막 분수','세는 방식','다른 분수'],
    hint:'힌트 — 분자와 분모의 합이 같은 것끼리 묶어 세면 됩니다.',
    read:()=>{ const n=+$('#ctN').value,row=$('#ctRow').checked,seq=order(n,row);
      const l=seq[seq.length-1];
      const seen=new Set(seq.map(([p,q])=>(p/gcd(p,q))+'/'+(q/gcd(p,q))));
      return [n,l?l[0]+'/'+l[1]:'—',row?'분모별':'대각선',seen.size]; } });
}

      return [n,'0.'+nw.slice(0,n).join(''),dup?'있음':'없음']; } });
}
