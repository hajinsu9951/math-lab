/* ---------- LAB 162 벡터의 합성 — 강을 건너는 배, 바람 속의 비행기 ---------- */
function vecAddDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="va1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">파란 화살표는 <b>배가 내는 힘</b>, 초록은 <b>물살</b>,
      붉은 화살표가 실제로 가는 방향입니다.</div>
    </div>
    <div class="col">
      ${slider('vaA','뱃머리 방향',0,180,5,90,'°')}
      ${slider('vaV','배의 속력',1,8,0.5,4,'m/s')}
      ${slider('vaC','물살의 속력',0,6,0.5,2,'m/s')}
      <div class="out" id="vaout"></div>
      <div class="note" style="margin-top:12px">두 벡터를 <b>이어 붙이면</b> 합이 됩니다.
      건너편 정면에 닿으려면 물살을 거슬러 각을 틀어야 합니다.</div>
    </div></div>`;
  const cv=$('#va1'),ctx=cv.getContext('2d'),W=380,H=320;
  function draw(){
    const a=+$('#vaA').value*Math.PI/180, v=+$('#vaV').value, c=+$('#vaC').value;
    $('#vaAv').textContent=$('#vaA').value+'°'; $('#vaVv').textContent=v.toFixed(1)+'m/s'; $('#vaCv').textContent=c.toFixed(1)+'m/s';
    const bx=v*Math.cos(a), by=v*Math.sin(a), rx=bx+c, ry=by;
    const X=x=>60+26*x, Y=y=>H-40-26*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='rgba(37,99,235,.06)'; ctx.fillRect(0,60,W,H-100);
    ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(0,Y(0)); ctx.lineTo(W,Y(0)); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(0,Y(8)); ctx.lineTo(W,Y(8)); ctx.stroke();
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('출발 기슭', 8, Y(0)+16); ctx.fillText('건너편 기슭 (폭 8 m)', 8, Y(8)-6);
    const arrow=(x1,y1,x2,y2,col,w)=>{ ctx.strokeStyle=col; ctx.fillStyle=col; ctx.lineWidth=w;
      ctx.beginPath(); ctx.moveTo(X(x1),Y(y1)); ctx.lineTo(X(x2),Y(y2)); ctx.stroke();
      const ang=Math.atan2(Y(y2)-Y(y1),X(x2)-X(x1));
      ctx.beginPath(); ctx.moveTo(X(x2),Y(y2));
      ctx.lineTo(X(x2)-9*Math.cos(ang-0.4),Y(y2)-9*Math.sin(ang-0.4));
      ctx.lineTo(X(x2)-9*Math.cos(ang+0.4),Y(y2)-9*Math.sin(ang+0.4));
      ctx.closePath(); ctx.fill(); };
    arrow(0,0,bx,by,'#2563EB',2.4);
    arrow(bx,by,rx,ry,'#0E9384',2.4);
    arrow(0,0,rx,ry,'#BE123C',3);
    ctx.setLineDash([4,4]); ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(0),Y(8)); ctx.stroke(); ctx.setLineDash([]);
    const t = ry>0.01 ? 8/ry : Infinity;
    const drift = isFinite(t) ? rx*t : Infinity;
    $('#vaout').innerHTML =
      `<span class="mono">배 (${bx.toFixed(2)}, ${by.toFixed(2)}) + 물살 (${c.toFixed(1)}, 0) = 합 <b>(${rx.toFixed(2)}, ${ry.toFixed(2)})</b></span><br>`
      + `<span class="mono">합의 크기 ${Math.hypot(rx,ry).toFixed(2)} m/s · 방향 ${(Math.atan2(ry,rx)*180/Math.PI).toFixed(1)}°</span><br>`
      + (isFinite(t) ? `<span class="mono">건너는 데 ${t.toFixed(1)}초 · 하류로 <b>${drift.toFixed(2)} m</b> 떠내려감</span><br>`
                     : `<b>강을 건너지 못합니다 — 세로 성분이 0 입니다.</b><br>`)
      + (Math.abs(drift)<0.15 ? `<b style="color:#B45309">정면 건너편에 그대로 닿습니다 — 물살을 정확히 상쇄했습니다.</b><br>` : '')
      + (c<0.05 ? `<b style="color:#7C3AED">물살이 없으니 뱃머리 방향이 곧 진행 방향입니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">두 화살표를 이어 붙인 끝이 실제 도착 방향입니다.</span>`;
  }
  ['vaA','vaV','vaC'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'va', title:'벡터의 합성 — 실제로 가는 방향',
    q:['뱃머리 각을 바꾸며 (각, 합벡터, 떠내려간 거리) 를 여섯 줄 기록하세요.',
       '떠내려가지 않으려면 각이 얼마여야 하나요? 물살이 세지면?',
       '물살이 배보다 빠르면 왜 정면에 닿을 수 없는지 설명하세요.'],
    cols:['뱃머리 각','배 속력','물살','합의 크기','떠내려감'],
    hint:'힌트 — 가로 성분이 0 이 되도록 cos 을 맞추면 됩니다.',
    read:()=>{ const a=+$('#vaA').value*Math.PI/180,v=+$('#vaV').value,c=+$('#vaC').value;
      const rx=v*Math.cos(a)+c, ry=v*Math.sin(a), t=ry>0.01?8/ry:Infinity;
      return [$('#vaA').value+'°',v.toFixed(1),c.toFixed(1),Math.hypot(rx,ry).toFixed(2),
        isFinite(t)?(rx*t).toFixed(2):'못 건넘']; } });
}

/* ---------- LAB 163 벡터의 실수배와 평행 조건 ---------- */
function vecScaleDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="vs1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">파란 화살표에 실수를 곱하면 <b>길이만</b> 변하거나
      <b>방향이 뒤집힙니다</b>. 초록 화살표와 나란해지는 순간을 찾아보세요.</div>
    </div>
    <div class="col">
      ${slider('vsK','곱하는 실수 k',-3,3,0.1,1,'')}
      ${slider('vsBX','비교할 벡터의 x',-6,6,0.5,3,'')}
      ${slider('vsBY','비교할 벡터의 y',-6,6,0.5,1,'')}
      <div class="out" id="vsout"></div>
      <div class="note" style="margin-top:12px">두 벡터가 평행 ⟺ 한쪽이 다른 쪽의 <b>실수배</b>
      ⟺ 성분의 <b>가위곱 a₁b₂ − a₂b₁ 이 0</b>.</div>
    </div></div>`;
  const cv=$('#vs1'),ctx=cv.getContext('2d'),W=380,H=320;
  const A=[2,1];
  function draw(){
    const k=+$('#vsK').value, bx=+$('#vsBX').value, by=+$('#vsBY').value;
    $('#vsKv').textContent=k.toFixed(1); $('#vsBXv').textContent=bx.toFixed(1); $('#vsBYv').textContent=by.toFixed(1);
    const X=x=>W/2+26*x, Y=y=>H/2-26*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.strokeStyle='#EEF1F5'; ctx.lineWidth=1;
    for(let g=-7;g<=7;g++){ ctx.beginPath(); ctx.moveTo(X(g),0); ctx.lineTo(X(g),H); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(0,Y(g)); ctx.lineTo(W,Y(g)); ctx.stroke(); }
    ctx.strokeStyle='#9CA3AF';
    ctx.beginPath(); ctx.moveTo(0,Y(0)); ctx.lineTo(W,Y(0)); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(X(0),0); ctx.lineTo(X(0),H); ctx.stroke();
    const arrow=(x,y,col,w)=>{ if(Math.hypot(x,y)<0.05) return;
      ctx.strokeStyle=col; ctx.fillStyle=col; ctx.lineWidth=w;
      ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(x),Y(y)); ctx.stroke();
      const ang=Math.atan2(Y(y)-Y(0),X(x)-X(0));
      ctx.beginPath(); ctx.moveTo(X(x),Y(y));
      ctx.lineTo(X(x)-10*Math.cos(ang-0.4),Y(y)-10*Math.sin(ang-0.4));
      ctx.lineTo(X(x)-10*Math.cos(ang+0.4),Y(y)-10*Math.sin(ang+0.4));
      ctx.closePath(); ctx.fill(); };
    ctx.strokeStyle='#C7D2E4'; ctx.lineWidth=1.2; ctx.setLineDash([4,4]);
    ctx.beginPath(); ctx.moveTo(X(-3*A[0]),Y(-3*A[1])); ctx.lineTo(X(3*A[0]),Y(3*A[1])); ctx.stroke(); ctx.setLineDash([]);
    arrow(A[0],A[1],'rgba(37,99,235,.35)',2);
    arrow(k*A[0],k*A[1],'#2563EB',3);
    arrow(bx,by,'#0E9384',2.6);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('a = (2, 1) · 진한 파랑 = k·a · 초록 = 비교 벡터', 12, H-12);
    const cross=A[0]*by-A[1]*bx;
    const need = A[0]!==0 ? bx/A[0] : by/A[1];
    $('#vsout').innerHTML =
      `<span class="mono">k·a = (${(k*A[0]).toFixed(1)}, ${(k*A[1]).toFixed(1)}) · 크기 ${Math.hypot(k*A[0],k*A[1]).toFixed(2)}</span><br>`
      + `<span class="mono">비교 벡터 b = (${bx.toFixed(1)}, ${by.toFixed(1)}) · a₁b₂ − a₂b₁ = <b>${cross.toFixed(2)}</b></span><br>`
      + (Math.abs(cross)<1e-9 ? `<b style="color:#B45309">가위곱이 0 — 두 벡터가 평행합니다 (b = ${need.toFixed(1)}·a).</b><br>` : '')
      + (k<0 ? `<b style="color:#7C3AED">k 가 음수라 화살표가 반대쪽을 가리킵니다 — 방향이 뒤집혔습니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">실수배는 회색 점선 위를 벗어나지 못합니다.</span>`;
  }
  ['vsK','vsBX','vsBY'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'vs', title:'실수배와 평행 조건',
    q:['k 를 바꾸며 (k, k·a 의 크기, 방향) 을 여섯 줄 기록하세요.',
       'b 를 a 와 평행하게 만드는 (x, y) 조합을 세 개 찾으세요.',
       '가위곱이 0 이라는 조건과 「한쪽이 다른 쪽의 실수배」가 같은 말임을 설명하세요.'],
    cols:['k','k·a','b','가위곱','평행?'],
    hint:'힌트 — a = (2, 1) 이므로 b = (2t, t) 꼴이면 평행합니다.',
    read:()=>{ const k=+$('#vsK').value,bx=+$('#vsBX').value,by=+$('#vsBY').value;
      return [k.toFixed(1),'('+(k*2).toFixed(1)+', '+(k*1).toFixed(1)+')','('+bx.toFixed(1)+', '+by.toFixed(1)+')',
        (2*by-1*bx).toFixed(2), Math.abs(2*by-bx)<1e-9?'평행':'아님']; } });
}

/* ---------- LAB 164 내적 — 두 도로는 몇 도로 만날까 ---------- */
function dotAngleDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="da1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">두 화살표의 <b>사잇각</b>을 내적으로 계산합니다.
      각을 90°로 만들면 내적이 어떻게 되는지 보세요.</div>
    </div>
    <div class="col">
      ${slider('daAX','a 의 x 성분',-6,6,0.5,4,'')}
      ${slider('daAY','a 의 y 성분',-6,6,0.5,1,'')}
      ${slider('daBX','b 의 x 성분',-6,6,0.5,2,'')}
      ${slider('daBY','b 의 y 성분',-6,6,0.5,4,'')}
      <div class="ctl" style="margin-top:8px"><button class="b g sm" id="daPerp">b 를 a 에 수직으로</button></div>
      <div class="out" id="daout"></div>
      <div class="note" style="margin-top:12px">a·b = |a||b|cosθ = a₁b₁ + a₂b₂.
      <b>내적이 0 이면 수직</b>, 음수면 둔각입니다.</div>
    </div></div>`;
  const cv=$('#da1'),ctx=cv.getContext('2d'),W=380,H=320;
  function draw(){
    const ax=+$('#daAX').value, ay=+$('#daAY').value, bx=+$('#daBX').value, by=+$('#daBY').value;
    ['daAX','daAY','daBX','daBY'].forEach(i=>$('#'+i+'v').textContent=(+$('#'+i).value).toFixed(1));
    const X=x=>W/2+26*x, Y=y=>H/2-26*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.strokeStyle='#EEF1F5'; ctx.lineWidth=1;
    for(let g=-7;g<=7;g++){ ctx.beginPath(); ctx.moveTo(X(g),0); ctx.lineTo(X(g),H); ctx.stroke();
      ctx.beginPath(); ctx.moveTo(0,Y(g)); ctx.lineTo(W,Y(g)); ctx.stroke(); }
    ctx.strokeStyle='#9CA3AF';
    ctx.beginPath(); ctx.moveTo(0,Y(0)); ctx.lineTo(W,Y(0)); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(X(0),0); ctx.lineTo(X(0),H); ctx.stroke();
    const arrow=(x,y,col)=>{ if(Math.hypot(x,y)<0.05) return;
      ctx.strokeStyle=col; ctx.fillStyle=col; ctx.lineWidth=3;
      ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(x),Y(y)); ctx.stroke();
      const ang=Math.atan2(Y(y)-Y(0),X(x)-X(0));
      ctx.beginPath(); ctx.moveTo(X(x),Y(y));
      ctx.lineTo(X(x)-10*Math.cos(ang-0.4),Y(y)-10*Math.sin(ang-0.4));
      ctx.lineTo(X(x)-10*Math.cos(ang+0.4),Y(y)-10*Math.sin(ang+0.4));
      ctx.closePath(); ctx.fill(); };
    arrow(ax,ay,'#2563EB'); arrow(bx,by,'#0E9384');
    const dot=ax*bx+ay*by, la=Math.hypot(ax,ay), lb=Math.hypot(bx,by);
    const cos=dot/Math.max(1e-9,la*lb), th=Math.acos(Math.max(-1,Math.min(1,cos)))*180/Math.PI;
    if(la>0.05&&lb>0.05){ ctx.strokeStyle='rgba(190,18,60,.8)'; ctx.lineWidth=1.6;
      ctx.beginPath(); ctx.arc(X(0),Y(0),26,-Math.atan2(ay,ax),-Math.atan2(by,bx),
        Math.atan2(ay,ax)<Math.atan2(by,bx)); ctx.stroke();
      ctx.fillStyle='#BE123C'; ctx.font='11px sans-serif'; ctx.textAlign='left';
      ctx.fillText(th.toFixed(1)+'°', X(0)+30, Y(0)-8); }
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('파랑 = a · 초록 = b', 12, H-12);
    $('#daout').innerHTML =
      `<span class="mono">a·b = ${ax.toFixed(1)}×${bx.toFixed(1)} + ${ay.toFixed(1)}×${by.toFixed(1)} = <b>${dot.toFixed(2)}</b></span><br>`
      + `<span class="mono">|a| = ${la.toFixed(2)} · |b| = ${lb.toFixed(2)} · cosθ = ${cos.toFixed(3)} · θ = <b>${th.toFixed(1)}°</b></span><br>`
      + (Math.abs(dot)<1e-9 ? `<b style="color:#B45309">내적이 0 — 두 벡터가 수직입니다 (θ = 90°).</b><br>` : '')
      + (dot<-1e-9 ? `<b style="color:#7C3AED">내적이 음수 — 둔각입니다. 서로 반대쪽을 향하는 성분이 있습니다.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">성분으로 곱해 더한 값과 |a||b|cosθ 가 같은지 확인하세요.</span>`;
  }
  $('#daPerp').onclick=()=>{ const ax=+$('#daAX').value, ay=+$('#daAY').value;
    $('#daBX').value=(-ay).toFixed(1); $('#daBY').value=ax.toFixed(1); draw(); };
  ['daAX','daAY','daBX','daBY'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'da', title:'내적 — 각을 재는 곱셈',
    q:['b 를 돌려 가며 (a·b, cosθ, θ) 를 여섯 줄 기록하세요.',
       '내적이 0, 양수, 음수일 때 각이 각각 어떤 범위인지 정리하세요.',
       '|a||b|cosθ 와 a₁b₁ + a₂b₂ 가 같은 값인지 두 방법으로 계산해 확인하세요.'],
    cols:['a','b','a·b','cosθ','θ'],
    hint:'힌트 — (x, y) 에 수직인 벡터는 (−y, x) 입니다.',
    read:()=>{ const ax=+$('#daAX').value,ay=+$('#daAY').value,bx=+$('#daBX').value,by=+$('#daBY').value;
      const dot=ax*bx+ay*by, la=Math.hypot(ax,ay), lb=Math.hypot(bx,by);
      const cos=dot/Math.max(1e-9,la*lb);
      return ['('+ax+', '+ay+')','('+bx+', '+by+')',dot.toFixed(2),cos.toFixed(3),
        (Math.acos(Math.max(-1,Math.min(1,cos)))*180/Math.PI).toFixed(1)+'°']; } });
}

/* ---------- LAB 165 정사영의 길이 — 비스듬한 그림자 ---------- */
function projLenDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="pl1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">막대를 기울이면 바닥의 <b>그림자</b>가 짧아집니다.
      각도와 그림자 길이의 관계를 보세요.</div>
    </div>
    <div class="col">
      ${slider('plL','막대의 길이',2,8,0.5,5,'m')}
      ${slider('plT','바닥과 이루는 각',0,90,5,30,'°')}
      <div class="out" id="plout"></div>
      <div class="note" style="margin-top:12px">정사영의 길이 = |a|cosθ.
      각이 <b>90°면 그림자가 한 점</b>, 0°면 길이 그대로입니다.</div>
    </div></div>`;
  const cv=$('#pl1'),ctx=cv.getContext('2d'),W=380,H=320;
  function draw(){
    const L=+$('#plL').value, t=+$('#plT').value*Math.PI/180;
    $('#plLv').textContent=L.toFixed(1)+'m'; $('#plTv').textContent=$('#plT').value+'°';
    const X=x=>50+34*x, Y=y=>H-70-34*y;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='rgba(107,114,128,.10)'; ctx.fillRect(0,Y(0),W,H-Y(0));
    ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1.2;
    ctx.beginPath(); ctx.moveTo(0,Y(0)); ctx.lineTo(W,Y(0)); ctx.stroke();
    const ex=L*Math.cos(t), ey=L*Math.sin(t);
    ctx.strokeStyle='#2563EB'; ctx.lineWidth=4; ctx.lineCap='round';
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)); ctx.lineTo(X(ex),Y(ey)); ctx.stroke(); ctx.lineCap='butt';
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=5;
    ctx.beginPath(); ctx.moveTo(X(0),Y(0)+8); ctx.lineTo(X(ex),Y(0)+8); ctx.stroke();
    ctx.setLineDash([4,4]); ctx.strokeStyle='#9CA3AF'; ctx.lineWidth=1;
    ctx.beginPath(); ctx.moveTo(X(ex),Y(ey)); ctx.lineTo(X(ex),Y(0)); ctx.stroke(); ctx.setLineDash([]);
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('파랑 = 막대 · 붉은 = 바닥에 생긴 그림자(정사영)', 12, H-12);
    ctx.fillStyle='#374151';
    ctx.fillText('그림자 '+ex.toFixed(2)+' m', X(ex/2)-24, Y(0)+30);
    const ratio=ex/L;
    $('#plout').innerHTML =
      `<span class="mono">|a| = ${L.toFixed(1)} · θ = ${$('#plT').value}° · 정사영 = |a|cosθ = <b style="color:#BE123C">${ex.toFixed(3)} m</b></span><br>`
      + `<span class="mono">줄어든 비율 = cos ${$('#plT').value}° = ${ratio.toFixed(3)}</span><br>`
      + (+$('#plT').value===90 ? `<b style="color:#B45309">막대가 수직 — 그림자가 한 점으로 줄었습니다 (cos 90° = 0).</b><br>` : '')
      + (+$('#plT').value===0 ? `<b style="color:#7C3AED">막대가 바닥에 붙어 그림자가 길이 그대로입니다 (cos 0° = 1).</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">그림자 길이는 각에 비례하지 않습니다 — cos 에 비례합니다.</span>`;
  }
  ['plL','plT'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'pl', title:'정사영의 길이 — cos 만큼 줄어든다',
    q:['각을 0°부터 90°까지 바꾸며 (각, 그림자, 비율) 을 여섯 줄 기록하세요.',
       '각이 두 배가 되면 그림자도 절반이 되나요? 표로 반박하세요.',
       '그림자가 원래 길이의 절반이 되는 각을 찾으세요.'],
    cols:['길이','각','cos','그림자','비율'],
    hint:'힌트 — 비율은 cos θ 입니다. 60°에서 얼마인가요?',
    read:()=>{ const L=+$('#plL').value,d=+$('#plT').value,t=d*Math.PI/180;
      return [L.toFixed(1),d+'°',Math.cos(t).toFixed(3),(L*Math.cos(t)).toFixed(3),Math.cos(t).toFixed(3)]; } });
}

/* ---------- LAB 166 정사영의 넓이 — 기울어진 판의 그림자 ---------- */
function projAreaDemo(root){
  root.innerHTML=`<div class="row"><div class="col">
      <canvas id="pa1" width="380" height="320"></canvas>
      <div class="note" style="margin-top:8px">기울어진 판을 바닥에서 올려다본 그림입니다.
      판을 세울수록 <b>그림자의 넓이</b>가 어떻게 되는지 보세요.</div>
    </div>
    <div class="col">
      ${slider('paW','판의 가로',1,6,0.5,4,'m')}
      ${slider('paH','판의 세로',1,6,0.5,3,'m')}
      ${slider('paT','바닥과 이루는 각',0,90,5,30,'°')}
      <div class="out" id="paout"></div>
      <div class="note" style="margin-top:12px">정사영의 넓이 S′ = S·cosθ.
      길이는 한 방향만 줄지만 <b>넓이도 cos 한 번</b>만 곱합니다 — 기울어진 방향으로만 줄기 때문입니다.</div>
    </div></div>`;
  const cv=$('#pa1'),ctx=cv.getContext('2d'),W=380,H=320;
  function draw(){
    const w=+$('#paW').value, h=+$('#paH').value, d=+$('#paT').value, t=d*Math.PI/180;
    $('#paWv').textContent=w.toFixed(1)+'m'; $('#paHv').textContent=h.toFixed(1)+'m'; $('#paTv').textContent=d+'°';
    const S=w*h, Sp=S*Math.cos(t);
    const sc=30, L=60, T=70;
    ctx.clearRect(0,0,W,H); ctx.fillStyle='#fff'; ctx.fillRect(0,0,W,H);
    ctx.fillStyle='rgba(37,99,235,.20)';
    ctx.beginPath();
    ctx.moveTo(L,T); ctx.lineTo(L+w*sc,T-w*sc*0.18); ctx.lineTo(L+w*sc,T-w*sc*0.18+h*sc*Math.cos(t));
    ctx.lineTo(L,T+h*sc*Math.cos(t)); ctx.closePath(); ctx.fill();
    ctx.strokeStyle='#2563EB'; ctx.lineWidth=2; ctx.stroke();
    ctx.fillStyle='rgba(190,18,60,.28)';
    ctx.fillRect(L, 210, w*sc, h*sc*Math.cos(t));
    ctx.strokeStyle='#BE123C'; ctx.lineWidth=1.6;
    ctx.strokeRect(L, 210, w*sc, h*sc*Math.cos(t));
    ctx.fillStyle='#6B7280'; ctx.font='10.5px sans-serif'; ctx.textAlign='left';
    ctx.fillText('파랑 = 기울어진 판 (실제 넓이 '+S.toFixed(1)+' m²)', 12, 40);
    ctx.fillText('붉은 = 바닥의 그림자 ('+Sp.toFixed(2)+' m²)', 12, 200);
    ctx.fillText('세로 방향만 cos θ 배로 줄어듭니다', 12, H-12);
    $('#paout').innerHTML =
      `<span class="mono">S = ${w.toFixed(1)} × ${h.toFixed(1)} = ${S.toFixed(1)} m² · θ = ${d}°</span><br>`
      + `<span class="mono">S′ = S·cosθ = ${S.toFixed(1)} × ${Math.cos(t).toFixed(3)} = <b style="color:#BE123C">${Sp.toFixed(2)} m²</b></span><br>`
      + (d===60 ? `<b style="color:#B45309">60° 로 세우니 그림자 넓이가 정확히 절반입니다 — cos 60° = 0.5.</b><br>` : '')
      + (d===90 ? `<b style="color:#7C3AED">판이 수직이 되어 그림자가 선분으로 줄었습니다 — 넓이 0.</b><br>` : '')
      + `<span style="color:#6B7280;font-size:12.5px">가로는 그대로, 세로만 줄어들기 때문에 cos 을 한 번만 곱합니다.</span>`;
  }
  ['paW','paH','paT'].forEach(i=>$('#'+i).oninput=draw); draw();
  inquiry(root,{ id:'pa', title:'정사영의 넓이 — S′ = S cos θ',
    q:['각을 바꾸며 (각, cos, 실제 넓이, 그림자 넓이) 를 여섯 줄 기록하세요.',
       '그림자가 절반이 되는 각을 찾고 그 이유를 쓰세요.',
       '왜 cos 을 두 번 곱하지 않는지 그림으로 설명하세요.'],
    cols:['가로','세로','각','S','S′'],
    hint:'힌트 — 기울어진 방향의 변만 줄어듭니다.',
    read:()=>{ const w=+$('#paW').value,h=+$('#paH').value,d=+$('#paT').value;
      return [w.toFixed(1),h.toFixed(1),d+'°',(w*h).toFixed(1),(w*h*Math.cos(d*Math.PI/180)).toFixed(2)]; } });
}
